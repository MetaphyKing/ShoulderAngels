==============================================================================
 ShoulderAngels — CHEAT SHEET
 Two voices debate your coding path forward (safe vs. bold + forecast)
==============================================================================

SETUP
  # Anthropic (default backend) — key required only on this path:
  export ANTHROPIC_API_KEY=sk-ant-...
  export ANTHROPIC_MODEL=claude-haiku-4-5    # optional, default shown

  # Agent / Grok host — no Anthropic key:
  export SHOULDERANGELS_BACKEND=command
  export SHOULDERANGELS_COMPLETE_CMD='python examples/fake_completer.py'

  pip install -e .                           # optional: installs `shoulderangels`

ONE-LINERS
  shoulderangels "Add caching to the user service"          # interactive (TTY)
  shoulderangels "Migrate to async" --choose bold --json    # pipeline-friendly
  shoulderangels "Refactor auth" --choose both --json       # forecast both paths
  shoulderangels task.md --choose safe                      # read task from file
  echo '...' | shoulderangels "Speed up build" --choose safe# never blocks on pipe
  shoulderangels --history                                  # list past decisions
  shoulderangels --history --json                           # full history as JSON
  shoulderangels "task" --choose safe --json \
    --backend command --complete-cmd "python examples/fake_completer.py"

FLAGS
  target              task string, or a path to a file describing the task
  --choose X          safe | bold | both   (skips the interactive prompt)
  --json              emit structured JSON
  --model ID          Anthropic model (default: $ANTHROPIC_MODEL or claude-haiku-4-5)
  --max-tokens N      cap tokens per Anthropic call (default 1024)
  --backend X         anthropic | command   (default: $SHOULDERANGELS_BACKEND or anthropic)
  --complete-cmd CMD  required for --backend command (or $SHOULDERANGELS_COMPLETE_CMD)
  --no-store          do not append this decision to history
  --history           print decision history and exit
  --version           print version

BEHAVIOR NOTES
  • Interactive prompt appears ONLY on a real TTY with no --choose given.
  • In pipes/CI with no --choose, it defaults to "safe" and never hangs.
  • History is stored at:  ~/.shoulderangels/history.json
  • All model output is JSON-validated; malformed responses degrade gracefully.
  • Agent backend = host LLM supplies complete(); Anthropic is optional.
  • Command completer: stdin {"prompt","system"} → stdout {"text":...} or raw.

EXIT CODES
  0  success
  1  LLM/key/network/command error (message on stderr)
  2  bad CLI usage (argparse)

TEST
  pytest -q            # 40 offline tests, LLM mocked, no API cost
